// create empty array
TriOsc array[4];

// for each, connect and set params
for( Osc x : array )
{
    // connect
    x => dac;
    // set gain
    .25 => x.gain;
    // set randomize freq
    Math.random2f(200,800) => x.freq;
}

// advance time
2::second => now;
